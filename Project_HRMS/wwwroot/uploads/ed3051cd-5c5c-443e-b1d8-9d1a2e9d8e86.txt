Create proc sp_FindSalary5
@Ename varchar(10),
@Esalary decimal(9,2)
as
begin
	
	declare @Epf decimal(9,2);
	declare @Enet decimal(9,2);

	set @Epf=@Esalary * 0.1;
	set @Enet = @Esalary - @Epf ;

	insert into Emp values(@Ename , @Esalary , @Epf , @Enet);


end
