Create proc sp_SaveEmp
@Ename varchar(10),
@Esalary decimal(9,2),
@mid int
as
begin
	
	declare @Epf decimal(9,2);
	declare @Enet decimal(9,2);

	set @Epf=@Esalary * 0.1;
	set @Enet = @Esalary - @Epf ;

	insert into emp values(@Ename , @Esalary , @Epf , @Enet,@mid);


end


create proc sp_Fetchall
as 
begin
	select e.eid,e.ename,e.esalary,e.pfamt,e.netsal,m.ename 
	from emp e inner join manager m on e.eid=m.mid;
end


create proc sp_Delete
@id int
as
begin
	declare @eid int
	select @eid=eid from emp;
	delete from emp where eid=@id;
end

create proc sp_SearchEmp
@name varchar(100)
as
begin
	select ename from emp where ename like @name;
end

create table empTask(
 eid int primary key identity,
 ename varchar(100),
 esalary decimal(9,2));

 select * from empTask;

alter proc sp_Insert
@name varchar(100),
@salary decimal(9,2)
as
begin

	declare @Ename varchar(100);
	select @Ename=ename from empTask where ename=@name;
	if(@Ename=@name)
	begin
		 Select ' Duplicate record' as Message;
	return;	
	end
	
	else
	begin
		insert into empTask (ename,esalary) values(@name,@salary)
		Select 'Emp add succesfully' as Message ;
	end
end


create proc sp_Fetch
as
begin
	select * from empTask;
end


alter proc sp_Update
@id int,
@name varchar(100),
@salary decimal(9,2)
as
begin
	declare @Eid int;
	select @Eid=eid from empTask;
	if(@Eid < @id)
	begin
		select 'ID not exists' as show
	return;
	end

	else
	begin
		update empTask set ename = @name, esalary=@salary 
		where eid=@id;
		select 'The data was updated' as show
	end
end

alter proc sp_signUp --Register
@usname varchar(100),
@email varchar(100),
@pass varchar(100),
@role varchar(100)
as
begin
	declare @Eusname varchar(100);
	declare @Eemail varchar(100);

	select @Eusname=username ,@Eemail=email from auth where email=@email OR username=@usname;

	if(@Eusname=@usname OR @Eemail=@email)
	begin
		select 'Username OR email already exist' as msg
	end

	else
	begin
	    insert into auth values(@usname,@email,@pass,@role);
	end
end
